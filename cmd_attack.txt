# ================= KỊCH BẢN TẤN CÔNG ==============================================

# ================= Tấn công UDP Flood - Tấn công băng thông

# Protocol: 17 (UDP), Packet/s tăng vọt.

h1 timeout 10s hping3 --flood --udp -p 80 h2

# -i u1000 nghĩa là interval 1000 microsecond = 1ms/gói = 1000 gói/s

# LOW RATE u10000-> 100pack/sec | High rate u100 -> 10000pack/sec

h1 hping3 --udp -p 80 -i u1000 h2

# 100 gói/giây - Kiểm tra độ nhạy của AI

h9 hping3 --udp -p 80 -i u1000 h10

# 10.000 gói/giây - Tải cao nhưng ko treo

h11 hping3 --udp -p 80 -i u100 h12

# ================= TCP SYN Flood - Tấn công tài nguyên kết nối

# TCP SYN - cờ flags, kích thước gói tin nhỏ, tần suất cao.

# Có ip_proto là 6, chỉ số pps cao

h3 timeout 20s hping3 -S --flood -p 80 h4
h3 timeout 20s hping3 -S -p 80 -i u1000 h4

# ================= ICMP Flood

h5 timeout 20s hping3 --flood --icmp h6
h5 timeout 20s hping3 --icmp -i u1000 h6

# DDoS - Distributed

# Controller sẽ hiển thị log cảnh báo với nhiều dòng Src khác nhau cùng trỏ về một Dst.

# Mở Server trên victim

mininet> xterm h18
iperf -s -u -p 80 &
